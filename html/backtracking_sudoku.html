Backtracking Algorithm Logic

Backtracking is a methodical way of trying out various sequences of decisions, until you find one that "works". In the context of Sudoku:

  1. Find an empty cell in the grid. If no empty cell is found, the puzzle is solved.
  2. Try placing digits 1 through 9 in that cell.
  3. For each digit, check if it is valid (does not violate Sudoku rules in the current row, column, and 3x3 subgrid).
  4. If a digit is valid, place it and recursively attempt to solve the rest of the board.
  5. If the placement leads to a solution, return true. If it leads to a dead end (no valid digits for the next cell), backtrack by removing the digit (resetting the cell to empty) and try the next digit.
#include <stdbool.h>

#define N 9

bool isSafe(int grid[N][N], int row, int col, int num) {
    for (int x = 0; x < N; x++) {
        if (grid[row][x] == num || grid[x][col] == num || 
            grid[row - row % 3 + x / 3][col - col % 3 + x % 3] == num) {
            return false;
        }
    }
    return true;
}

bool solveSudoku(int grid[N][N]) {
    int row = -1, col = -1;
    bool isEmpty = false;
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) {
            if (grid[i][j] == 0) {
                row = i; col = j; isEmpty = true; break;
            }
        }
        if (isEmpty) break;
    }

    if (!isEmpty) return true;

    for (int num = 1; num <= 9; num++) {
        if (isSafe(grid, row, col, num)) {
            grid[row][col] = num;
            if (solveSudoku(grid)) return true;
            grid[row][col] = 0; // Backtrack
        }
    }
    return false;
}

class Sudoku {
    public boolean isSafe(int[][] board, int row, int col, int num) {
        for (int i = 0; i < 9; i++) {
            if (board[i][col] != 0 && board[i][col] == num) return false;
            if (board[row][i] != 0 && board[row][i] == num) return false;
            if (board[3 * (row / 3) + i / 3][3 * (col / 3) + i % 3] != 0 && 
                board[3 * (row / 3) + i / 3][3 * (col / 3) + i % 3] == num) {
                return false;
            }
        }
        return true;
    }

    public boolean solveSudoku(int[][] board) {
        for (int i = 0; i < 9; i++) {
            for (int j = 0; j < 9; j++) {
                if (board[i][j] == 0) {
                    for (int num = 1; num <= 9; num++) {
                        if (isSafe(board, i, j, num)) {
                            board[i][j] = num;
                            if (solveSudoku(board)) return true;
                            board[i][j] = 0; // Backtrack
                        }
                    }
                    return false;
                }
            }
        }
        return true;
    }
}

def is_safe(board, row, col, num):
    for i in range(9):
        if board[row][i] == num or board[i][col] == num:
            return False
            
    start_row = row - row % 3
    start_col = col - col % 3
    for i in range(3):
        for j in range(3):
            if board[i + start_row][j + start_col] == num:
                return False
    return True

def solve_sudoku(board):
    for i in range(9):
        for j in range(9):
            if board[i][j] == 0:
                for num in range(1, 10):
                    if is_safe(board, i, j, num):
                        board[i][j] = num
                        if solve_sudoku(board):
                            return True
                        board[i][j] = 0 # Backtrack
                return False
    return True
