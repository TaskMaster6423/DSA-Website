Prim's Algorithm Logic

Prim's Algorithm finds the Minimum Spanning Tree (MST) of a weighted undirected graph by building it one vertex at a time, from an arbitrary starting vertex.

  1. Initialize all vertices with a key of Infinity, except the starting vertex which gets a key of 0.
  2. While there are vertices not yet included in the MST:
    • Extract the vertex with the minimum key value that is not in the MST.
    • Include this vertex in the MST.
    • For each neighbor of this vertex, if the neighbor is not in the MST and the edge weight is less than its current key value, update its key to this edge weight.
#include <stdio.h>
#include <limits.h>
#include <stdbool.h>

#define V 5

int minKey(int key[], bool mstSet[]) {
    int min = INT_MAX, min_index;

    for (int v = 0; v < V; v++)
        if (mstSet[v] == false && key[v] < min)
            min = key[v], min_index = v;

    return min_index;
}

void primMST(int graph[V][V]) {
    int parent[V]; 
    int key[V];    
    bool mstSet[V]; 

    for (int i = 0; i < V; i++)
        key[i] = INT_MAX, mstSet[i] = false;

    key[0] = 0;
    parent[0] = -1;

    for (int count = 0; count < V - 1; count++) {
        int u = minKey(key, mstSet);
        mstSet[u] = true;

        for (int v = 0; v < V; v++)
            if (graph[u][v] && mstSet[v] == false && graph[u][v] < key[v])
                parent[v] = u, key[v] = graph[u][v];
    }
}

#include <iostream>
#include <vector>
#include <queue>
using namespace std;

typedef pair<int, int> pii;

void primMST(int V, vector<vector<pii>>& adj) {
    priority_queue<pii, vector<pii>, greater<pii>> pq;
    vector<int> key(V, 1e9);
    vector<int> parent(V, -1);
    vector<bool> inMST(V, false);

    pq.push({0, 0});
    key[0] = 0;

    while (!pq.empty()) {
        int u = pq.top().second;
        pq.pop();

        if (inMST[u]) continue;
        inMST[u] = true;

        for (auto x : adj[u]) {
            int v = x.first;
            int weight = x.second;

            if (!inMST[v] && key[v] > weight) {
                key[v] = weight;
                pq.push({key[v], v});
                parent[v] = u;
            }
        }
    }
}

import java.util.*;

class Prim {
    static class Edge {
        int to, weight;
        Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    public static void primMST(int V, List<List<Edge>> adj) {
        boolean[] inMST = new boolean[V];
        int[] key = new int[V];
        int[] parent = new int[V];
        Arrays.fill(key, Integer.MAX_VALUE);
        
        PriorityQueue<Edge> pq = new PriorityQueue<>(Comparator.comparingInt(e -> e.weight));
        
        key[0] = 0;
        pq.add(new Edge(0, 0));
        
        while (!pq.isEmpty()) {
            int u = pq.poll().to;
            
            if (inMST[u]) continue;
            inMST[u] = true;
            
            for (Edge edge : adj.get(u)) {
                int v = edge.to;
                int weight = edge.weight;
                
                if (!inMST[v] && key[v] > weight) {
                    key[v] = weight;
                    pq.add(new Edge(v, key[v]));
                    parent[v] = u;
                }
            }
        }
    }
}

def prim_mst(V, adj):
    key = [float('inf')] * V
    parent = [-1] * V
    in_mst = [False] * V
    
    pq = [(0, 0)]
    key[0] = 0
    
    while pq:
        weight, u = heapq.heappop(pq)

            continue

            if not in_mst[v] and key[v] > w:
                key[v] = w
                heapq.heappush(pq, (key[v], v))
