Dijkstra's Algorithm Logic

Dijkstra's Algorithm finds the shortest path from a starting node to all other nodes in a weighted graph with non-negative weights.

  1. Initialize the distance to the Start node as 0 and to all other nodes as Infinity.
  2. Create a Priority Queue (or min-heap) to continually extract the unvisited node with the smallest distance.
  3. While the queue is not empty:
    • Extract the node u with the minimum distance.
    • For each neighboring node v, perform Relaxation:
    • Calculate tentative_dist = dist[u] + weight(u, v).
    • If tentative_dist < dist[v], update dist[v] and push v to the queue.

}

#include <iostream>
#include <vector>
#include <queue>
using namespace std;

const int INF = 1e9;

void dijkstra(int start, int V, vector<vector<pair<int, int>>>& adj) {
    vector<int> dist(V, INF);
    dist[start] = 0;
    
    // priority_queue<distance, node>
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    pq.push({0, start});

    while (!pq.empty()) {
        int d = pq.top().first;
        int u = pq.top().second;
        pq.pop();

        if (d > dist[u]) continue;

        for (auto& edge : adj[u]) {
            int v = edge.first;
            int weight = edge.second;

            if (dist[u] + weight < dist[v]) {
                dist[v] = dist[u] + weight;
                pq.push({dist[v], v});
            }
        }
    }
}

import java.util.*;

class Dijkstra {
    static class Edge {
        int target, weight;
        Edge(int target, int weight) { this.target = target; this.weight = weight; }
    }

    public static void dijkstra(int start, int V, List<List<Edge>> adj) {
        int[] dist = new int[V];
        Arrays.fill(dist, Integer.MAX_VALUE);
        dist[start] = 0;

        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> a[0] - b[0]);
        pq.add(new int[]{0, start});

        while (!pq.isEmpty()) {
            int[] current = pq.poll();
            int d = current[0];
            int u = current[1];

            if (d > dist[u]) continue;

            for (Edge edge : adj.get(u)) {
                if (dist[u] + edge.weight < dist[edge.target]) {
                    dist[edge.target] = dist[u] + edge.weight;
                    pq.add(new int[]{dist[edge.target], edge.target});
                }
            }
        }
    }
}

import heapq

def dijkstra(start, V, adj):
    dist = [float('inf')] * V
    dist[start] = 0
    
    # Priority queue stores tuples of (distance, node)
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        
        if d > dist[u]:
            continue
            
        for v, weight in adj[u]:
            if dist[u] + weight < dist[v]:
                dist[v] = dist[u] + weight
                heapq.heappush(pq, (dist[v], v))
