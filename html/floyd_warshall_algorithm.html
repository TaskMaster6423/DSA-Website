Floyd-Warshall Algorithm Logic

The Floyd-Warshall algorithm computes shortest paths between all pairs of vertices using Dynamic Programming.

  1. Initialize the solution matrix same as the input graph matrix as a first step.
  2. Pick an intermediate vertex `k` from `0` to `V-1`.
  3. For every pair `(i, j)`, check if going through intermediate vertex `k` yields a shorter path: `dist[i][k] + dist[k][j] < dist[i][j]`.
  4. If true, update `dist[i][j] = dist[i][k] + dist[k][j]`.
#include <stdio.h>
#define INF 99999
#define V 4

void floydWarshall(int dist[][V]) {
    for (int k = 0; k < V; k++) {
        for (int i = 0; i < V; i++) {
            for (int j = 0; j < V; j++) {
                if (dist[i][k] + dist[k][j] < dist[i][j])
                    dist[i][j] = dist[i][k] + dist[k][j];
            }
        }
    }
}

#include <vector>
#include <algorithm>
using namespace std;
const int INF = 1e9;

void floydWarshall(vector<vector<int>>& dist) {
    int V = dist.size();
    for (int k = 0; k < V; k++) {
        for (int i = 0; i < V; i++) {
            for (int j = 0; j < V; j++) {
                if (dist[i][k] < INF && dist[k][j] < INF) {
                    dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);
                }
            }
        }
    }
}
