A* Algorithm Logic

A* Search Algorithm finds the shortest path from a starting node to a target node by combining the actual cost from the start (G-cost) and the estimated cost to the goal (Heuristic or H-cost).

  1. Initialize the Open Set (priority queue) with the start node, setting its G-cost to 0 and F-cost to the heuristic distance to the end node.
  2. While the Open Set is not empty:
    • Extract the node with the lowest F-cost (F = G + H).
    • If this node is the target, the shortest path has been found.
    • Otherwise, move the node to the Closed Set.
    • For each neighbor, calculate its tentative G-cost. If it is lower than the previously recorded G-cost, update its G-cost, recalculate its F-cost, and add it to the Open Set.

#include <iostream>
#include <vector>
#include <queue>
#include <cmath>
using namespace std;

struct Node { int x, y; };
struct Edge { int to, weight; };

int heuristic(Node a, Node b) {
    return sqrt(pow(a.x - b.x, 2) + pow(a.y - b.y, 2));
}

void aStar(int start, int end, vector<Node>& nodes, vector<vector<Edge>>& adj) {
    int n = nodes.size();
    vector<int> gScore(n, 1e9);
    vector<int> fScore(n, 1e9);
    
    // priority_queue stores {fScore, nodeIndex}
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> openSet;
    
    gScore[start] = 0;
    fScore[start] = heuristic(nodes[start], nodes[end]);
    openSet.push({fScore[start], start});
    
    while(!openSet.empty()) {
        int current = openSet.top().second;
        openSet.pop();
        
        if (current == end) return; // Path found
        
        for (auto& edge : adj[current]) {
            int neighbor = edge.to;
            int tentativeG = gScore[current] + edge.weight;
            
            if (tentativeG < gScore[neighbor]) {
                gScore[neighbor] = tentativeG;
                fScore[neighbor] = tentativeG + heuristic(nodes[neighbor], nodes[end]);
                openSet.push({fScore[neighbor], neighbor});
            }
        }
    }
}

class AStar {
    static class Node { int x, y; }
    static class Edge { int to, weight; Edge(int t, int w) { to=t; weight=w; } }
    
    static int heuristic(Node a, Node b) {
        return (int)Math.sqrt(Math.pow(a.x - b.x, 2) + Math.pow(a.y - b.y, 2));
    }
    
    public static void aStar(int start, int end, Node[] nodes, List<List<Edge>> adj) {
        int n = nodes.length;
        int[] gScore = new int[n];
        int[] fScore = new int[n];
        Arrays.fill(gScore, Integer.MAX_VALUE);
        Arrays.fill(fScore, Integer.MAX_VALUE);
        
        PriorityQueue<int[]> openSet = new PriorityQueue<>((a, b) -> a[0] - b[0]);
        
        gScore[start] = 0;
        fScore[start] = heuristic(nodes[start], nodes[end]);
        openSet.add(new int[]{fScore[start], start});
        
        while(!openSet.isEmpty()) {
            int current = openSet.poll()[1];
            
            if(current == end) return; // Path found
            
            for(Edge edge : adj.get(current)) {
                int neighbor = edge.to;
                int tentativeG = gScore[current] + edge.weight;
                
                if(tentativeG < gScore[neighbor]) {
                    gScore[neighbor] = tentativeG;
                    fScore[neighbor] = tentativeG + heuristic(nodes[neighbor], nodes[end]);
                    openSet.add(new int[]{fScore[neighbor], neighbor});
                }
            }
        }
    }
}
