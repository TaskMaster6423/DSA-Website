Kruskal's Algorithm Logic

Kruskal's Algorithm finds the Minimum Spanning Tree (MST) of a weighted undirected graph by sorting all edges and adding them one by one, provided they don't form a cycle.

  1. Sort all edges in non-decreasing order of their weight.
  2. Pick the smallest edge. Check if it forms a cycle with the spanning tree formed so far. This is typically done using a Disjoint Set (Union-Find) data structure.
  3. If the cycle is not formed, include this edge. Otherwise, discard it.
  4. Repeat step 2 until there are (V - 1) edges in the spanning tree, where V is the number of vertices.
#include <stdio.h>
#include <stdlib.h>

struct Edge { int u, v, weight; };

int compareEdges(const void* a, const void* b) {
    return ((struct Edge*)a)->weight - ((struct Edge*)b)->weight;
}

int find(int i, int parent[]) {
    if (parent[i] == i) return i;
    return parent[i] = find(parent[i], parent); // Path compression
}

void unionSets(int i, int j, int parent[]) {
    int root_i = find(i, parent);
    int root_j = find(j, parent);
    if (root_i != root_j) parent[root_i] = root_j;
}

void kruskal(int V, int E, struct Edge edges[]) {
    qsort(edges, E, sizeof(struct Edge), compareEdges);

    int parent[V];
    for (int i = 0; i < V; i++) parent[i] = i;

    int mstWeight = 0;
    for (int i = 0; i < E; i++) {
        int u = edges[i].u;
        int v = edges[i].v;
        int root_u = find(u, parent);
        int root_v = find(v, parent);

        if (root_u != root_v) {
            unionSets(root_u, root_v, parent);
            mstWeight += edges[i].weight;
        }
    }
}

void unionSets(int i, int j, vector<int>& parent) {
    int root_i = find(i, parent);
    int root_j = find(j, parent);
    if (root_i != root_j) parent[root_i] = root_j;
}

void kruskal(int V, vector<Edge>& edges) {
    sort(edges.begin(), edges.end(), compareEdges);

    vector<int> parent(V);
    for (int i = 0; i < V; i++) parent[i] = i;

    for (Edge e : edges) {
        int root_u = find(e.u, parent);
        int root_v = find(e.v, parent);

        if (root_u != root_v) {
            unionSets(root_u, root_v, parent);
            // Edge e is part of MST
        }
    }
}

import java.util.*;

class Kruskal {
    static class Edge implements Comparable<Edge> {
        int src, dest, weight;
        public int compareTo(Edge e) { return this.weight - e.weight; }
    }

    static int find(int[] parent, int i) {
        if (parent[i] == i) return i;
        return parent[i] = find(parent, parent[i]);
    }

    static void union(int[] parent, int i, int j) {
        int root1 = find(parent, i);
        int root2 = find(parent, j);
        if (root1 != root2) parent[root1] = root2;
    }

    public static void kruskal(int V, List<Edge> edges) {
        Collections.sort(edges);
        int[] parent = new int[V];
        for (int i = 0; i < V; i++) parent[i] = i;

        for (Edge e : edges) {
            int root1 = find(parent, e.src);
            int root2 = find(parent, e.dest);
            if (root1 != root2) {
                union(parent, root1, root2);
                // Edge e is part of MST
            }
        }
    }
}

def kruskal(vertices, edges):
    edges.sort(key=lambda x: x[2]) # Sort by weight
    parent = list(range(vertices)) # Make sets

    def find(i):
        if parent[i] == i: return i
        parent[i] = find(parent[i])
        return parent[i]

    def union(i, j):
        root_i = find(i)
        root_j = find(j)
        if root_i != root_j: 
            parent[root_i] = root_j

    mst = []
