Radix Sort Logic

  1. Digit by Digit: Radix sort processes numbers digit by digit, starting from the least significant digit (LSD) to the most significant digit (MSD).
  2. Distribution (Buckets): Elements are placed into 10 buckets (0-9) based on their current digit.
  3. Collection: Elements are collected back from the buckets in order, preserving the relative order from the previous steps.
  4. Repeat: This process is repeated for each digit position until the maximum number is fully processed.

void countSort(int arr[], int n, int exp) {
    int output[n];
    int i, count[10] = {0};

    for (i = 0; i < n; i++)
        count[(arr[i] / exp) % 10]++;

    for (i = 1; i < 10; i++)
        count[i] += count[i - 1];

    for (i = n - 1; i >= 0; i--) {
        output[count[(arr[i] / exp) % 10] - 1] = arr[i];
        count[(arr[i] / exp) % 10]--;
    }

    for (i = 0; i < n; i++)
        arr[i] = output[i];
}

def countSort(arr, exp):
    n = len(arr)
    output = [0] * n
    count = [0] * 10

    for i in range(n):
        index = arr[i] // exp
        count[index % 10] += 1

    for i in range(1, 10):
        count[i] += count[i - 1]

    i = n - 1
    while i >= 0:
        index = arr[i] // exp
        output[count[index % 10] - 1] = arr[i]
        count[index % 10] -= 1
        i -= 1

    for i in range(n):
        arr[i] = output[i]

def radixSort(arr):
    if not arr: return
    max1 = max(arr)
    exp = 1
    while max1 // exp > 0:
        countSort(arr, exp)
        exp *= 10
